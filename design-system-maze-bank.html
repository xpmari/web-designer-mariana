05 — Sombras & Efeitos

Tokens Completos

Variáveis CSS prontas para colar

Basta copiar o bloco abaixo e incluir no topo do seu arquivo global de estilos.

CSS :root — Maze Bank Design System (Rosa #FF69B4) 
/* ============================================================
   MAZE BANK — DESIGN SYSTEM TOKENS (ROSA #FF69B4)
   ============================================================ */

:root {
  /* ── Fundos & Vidro ─────────────────────────────────────── */
  --bg-base:             #030712;                   /* gray-950 */
  --surface-default:     #111827;                   /* gray-900 */
  --surface-panel:       rgba(17, 24, 39, 0.80);    /* painel com blur */
  --glass-default:       rgba(255, 255, 255, 0.05); /* cards / inputs */
  --glass-hover:         rgba(255, 255, 255, 0.10); /* estados de hover */

  /* ── Tipografia & Cores de Texto ────────────────────────── */
  --text-primary:        #F1F5F9;                   /* slate-100 */
  --text-muted:          #CBD5E1;                   /* slate-300 */
  --text-dim:            #94A3B8;                   /* slate-400 */
  --font-family:         'Inter', ui-sans-serif, system-ui, sans-serif;

  /* ── Identidade da Marca (Rosa #FF69B4 / Gradiente) ─────── */
  --brand-pink:          #FF69B4;                   /* Cor Principal */
  --brand-pink-light:    #FF8DC7;                   /* Destaque suave */
  --brand-pink-deep:     #FF1493;                   /* Rosa profundo */
  --brand-rose:          #F43F5E;                   /* Rose-500 */
  --brand-gradient:      linear-gradient(to right, #FF69B4, #FF1493, #F43F5E);

  /* ── Status Semântico ───────────────────────────────────── */
  --status-emerald:      #34D399;                   /* emerald-400 */
  --status-emerald-bg:   rgba(16, 185, 129, 0.15);  /* badge positivo */

  /* ── Bordas & Sombras ───────────────────────────────────── */
  --border-subtle:       rgba(255, 255, 255, 0.10); /* ring-white/10 */
  --glow-primary:        0 10px 15px -3px rgba(255, 105, 180, 0.35);
  --shadow-panel-deep:   0 25px 50px -12px rgba(255, 20, 147, 0.25);
}

/* ── Assinatura: Borda de Vidro Lapidado ──────────────────── */
.border-gradient {
  position: relative;
}
.border-gradient::before {
  content: "";
  position: absolute;
  inset: 0;
  border-radius: inherit;
  padding: 1px;
  -webkit-mask: linear-gradient(#fff 0 0) content-box, linear-gradient(#fff 0 0);
  -webkit-mask-composite: xor;
  mask-composite: exclude;
  background: linear-gradient(225deg,
    rgba(255, 255, 255, 0.0) 0%,
    rgba(255, 255, 255, 0.2) 50%,
    rgba(255, 255, 255, 0.0) 100%);
  pointer-events: none;
}
